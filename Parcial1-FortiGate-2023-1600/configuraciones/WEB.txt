# /etc/network/interfaces
auto lo
iface lo inet loopback

auto eth0
iface eth0 inet static
    address 172.23.16.130
    netmask 255.255.255.240
    gateway 172.23.16.129

# Instalacion
apk add nginx
mkdir -p /var/www/html/inventario
echo '<h1>WEB Server - Abdul Djalo 2023-1600</h1>' > /var/www/html/index.html
echo '<h1>INVENTARIO - area restringida</h1>' > /var/www/html/inventario/index.html

# /etc/nginx/http.d/default.conf
server {
    listen 80 default_server;
    server_name _;
    root /var/www/html;
    index index.html;
    location /inventario/ {
        autoindex on;
    }
}

rc-update add nginx default
rc-service nginx start
