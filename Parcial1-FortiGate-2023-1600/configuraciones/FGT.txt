# --- Interfaces ---
config system interface
    edit "port1"
        set alias "WAN"
        set ip 200.23.16.2 255.255.255.252
        set allowaccess ping http https ssh
        set role wan
    next
    edit "VLAN10"
        set vdom "root"
        set ip 172.23.16.1 255.255.255.128
        set allowaccess ping
        set role lan
        set interface "port2"
        set vlanid 10
    next
    edit "VLAN20"
        set ip 172.23.16.161 255.255.255.224
        set allowaccess ping
        set role lan
        set interface "port2"
        set vlanid 20
    next
    edit "VLAN30"
        set ip 172.23.16.129 255.255.255.240
        set allowaccess ping
        set role lan
        set interface "port2"
        set vlanid 30
    next
end

# --- Zona LAN ---
config system zone
    edit "LAN"
        set interface "VLAN10" "VLAN20" "VLAN30"
    next
end

# --- Servidor DHCP (VLAN 10 y VLAN 20) ---
config system dhcp server
    edit 1
        set default-gateway 172.23.16.1
        set netmask 255.255.255.128
        set interface "VLAN10"
        config ip-range
            edit 1
                set start-ip 172.23.16.10
                set end-ip 172.23.16.120
            next
        end
        set dns-server1 8.8.8.8
    next
    edit 2
        set default-gateway 172.23.16.161
        set netmask 255.255.255.224
        set interface "VLAN20"
        config ip-range
            edit 1
                set start-ip 172.23.16.162
                set end-ip 172.23.16.180
            next
        end
        set dns-server1 8.8.8.8
    next
end

# --- Ruta por defecto ---
config router static
    edit 1
        set gateway 200.23.16.1
        set device "port1"
    next
end

# --- Perfil Web Filter: bloquea /inventario (requisito 2) ---
config webfilter profile
    edit "WF-Inventario"
        config web
            set urlfilter-table 1
        end
    next
end
config webfilter urlfilter
    edit 1
        set name "Bloqueo-Inventario"
        config entries
            edit 1
                set url "*inventario*"
                set type wildcard
                set action block
            next
        end
    next
end

# --- Politicas de firewall ---
config firewall policy
    edit 1
        set name "LAN-a-Internet"
        set srcintf "LAN"
        set dstintf "port1"
        set srcaddr "all"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
        set nat enable
        set logtraffic all
    next
    edit 2
        set name "SSH-VLAN20-OK"
        set srcintf "LAN"
        set dstintf "LAN"
        set srcaddr "VLAN20 address"
        set dstaddr "VLAN30 address"
        set action accept
        set schedule "always"
        set service "SSH"
        set logtraffic all
    next
    edit 3
        set name "SSH-VLAN10-DENY"
        set srcintf "LAN"
        set dstintf "LAN"
        set srcaddr "VLAN10 address"
        set dstaddr "VLAN30 address"
        set action deny
        set schedule "always"
        set service "SSH"
        set logtraffic all
    next
    edit 4
        set name "VLAN10-a-Servidores-WEB"
        set srcintf "LAN"
        set dstintf "LAN"
        set srcaddr "VLAN10 address"
        set dstaddr "VLAN30 address"
        set action accept
        set schedule "always"
        set service "HTTP" "HTTPS" "PING"
        set utm-status enable
        set webfilter-profile "WF-Inventario"
        set logtraffic all
    next
end
